async function orgCategoryMembers(category,deep=false,base='https://marvel.fandom.com'){
 const all=new Map(),queue=[category],visited=new Set();
 while(queue.length){
  const current=queue.shift();if(visited.has(current))continue;visited.add(current);
  let cont=null;
  do{
   const p=new URLSearchParams({action:'query',list:'categorymembers',cmtitle:current,cmlimit:'500',cmtype:'page|subcat',format:'json',origin:'*'});
   if(cont)Object.entries(cont).forEach(([k,v])=>p.set(k,v));
   const r=await fetch(base+'/api.php?'+p.toString(),{headers:{Accept:'application/json'}});
   if(!r.ok)throw new Error('Source returned HTTP '+r.status+' while reading '+current);
   const j=await r.json();if(j.error)throw new Error(j.error.info||'Source API error');
   for(const x of (j.query?.categorymembers||[])){
    if(x.ns===0)all.set(norm(x.title),{name:x.title,pageid:x.pageid,source:base.replace('https://','')+' · '+current});
    else if(deep&&x.ns===14&&!visited.has(x.title))queue.push(x.title);
   }
   cont=j.continue||null;
  }while(cont);
 }
 return [...all.values()];
}

function hexToRgb(h){const n=parseInt(h.slice(1),16);return [(n>>16)&255,(n>>8)&255,n&255]}

function mixHex(a,b,t){
  if(t<=0)return a;
  const k=a+'|'+b+'|'+t;let v=colCache.get(k);
  if(!v){const A=hexToRgb(a),B=hexToRgb(b);v='rgb('+A.map((q,i)=>Math.round(q+(B[i]-q)*t)).join(',')+')';colCache.set(k,v)}

  return v;
}

function toonWindows(x,cx,top,w,sw,h,d,o,wins,arch){
  const front=Math.max(14,w-sw-7),right=cx+front,bottom=top+h-4,id=arch&&arch.id||'',facade=({ 'manhattan-glass-super-tower':'glass-bands','slender-tapered-skyscraper':'vertical-bays','curved-corner-curtain-wall':'glass-bands','art-deco-setback-tower':'masonry-bands','prewar-limestone-apartment':'masonry-bands','brownstone-residential-block':'balconies','terraced-residential-highrise':'masonry-bands','industrial-loft-building':'loft-bays','civic-cultural-complex':'masonry-bands','modern-office-midrise':'glass-bands','waterfront-warehouse':'industrial' })[id]||(arch&&arch.facade)||'curtain-wall';
  const palette=CITY_ARCH_CONFIG&&CITY_ARCH_CONFIG.palette||{};
  const glass=palette.glass||['#9FC8D3','#B8D3D4','#789EAF','#D3D9CF'];
  const warm=palette.warmInterior||['#F4D8A4','#EBC28A','#D8A77A'];
  const masonry=['masonry-bands','balconies','loft-bays','industrial'].includes(facade);
  const floors=Math.max(5,Math.floor(h/(masonry?19:25)));
  const bays=Math.max(3,Math.floor(front/(facade==='glass-bands'?17:facade==='vertical-bays'?15:21)));
  const bay=front/bays;
  x.save();
  /* Deep shadowed facade cavity: the glass sits behind the structural frame. */
  const bed=x.createLinearGradient(cx,top,right,bottom);
  bed.addColorStop(0,'rgba(204,231,237,.34)');
  bed.addColorStop(.24,'rgba(117,166,183,.38)');
  bed.addColorStop(.68,'rgba(53,91,111,.46)');
  bed.addColorStop(1,'rgba(22,48,67,.62)');

  for(let col=0;col<bays;col++){
    const bx=cx+col*bay,bw=bay;

    for(let row=0;row<floors;row++){
      const fh=(h-10)/floors,yy=top+5+row*fh+stagger;
      if(yy+fh>bottom)continue;
      const seed=(col*7+row*11+(Math.floor(cx)*3))%13;
      if(masonry){
        const ww=bw*(.48+((seed%5)*.055)),wh=Math.max(6,fh*(.48+((seed%4)*.075)));

        x.fillStyle='rgba(245,250,245,.32)';x.fillRect(xx+1,wy+1,Math.max(1,ww*.12),wh-2);
        x.strokeStyle='rgba(24,47,62,.68)';x.lineWidth=.65;
        x.beginPath();x.moveTo(xx+ww*.5,wy);x.lineTo(xx+ww*.5,wy+wh);x.stroke();
        if(facade==='balconies'&&seed%4===0){
          x.fillStyle='rgba(30,55,68,.65)';x.fillRect(xx-2,wy+wh+1,ww+4,1.8);
          x.strokeStyle='rgba(225,239,239,.58)';x.lineWidth=.55;
          x.beginPath();x.moveTo(xx-1,wy+wh+2);x.lineTo(xx+ww+1,wy+wh+2);x.stroke();
        }
      }else{
        const pad=facade==='atrium-grid'?1.8:2.2;
        const ww=Math.max(3,bw-pad*2),wh=Math.max(4,fh-(seed%4===0?1.8:3.1));
        x.fillStyle='rgba(10,27,42,.62)';x.fillRect(bx+pad-.8,yy-.5,ww+1.6,wh+1);
        const gc=pick(glass),g=x.createLinearGradient(bx,yy,bx+bw,yy+wh);
        g.addColorStop(0,shadeOf(gc,1.18));g.addColorStop(.42,gc);g.addColorStop(1,'rgba(27,58,78,.9)');
        x.fillStyle=g;x.fillRect(bx+pad,yy,ww,wh);
        /* Occasional warm rooms, never a uniform checkerboard. */
        if(seed%17===0){x.fillStyle=pick(warm);x.globalAlpha=.62;x.fillRect(bx+pad+1,yy+1,ww-2,wh-2);x.globalAlpha=1}
        /* Narrow reflected sky strips and angled glass highlights. */
        if(seed%3!==0){
          x.fillStyle='rgba(239,250,248,.17)';x.fillRect(bx+pad+1,yy+1,Math.max(.8,ww*(.08+(seed%4)*.025)),wh-2);
        }
        if(seed%7===0){

          x.beginPath();x.moveTo(bx+pad+ww*.22,yy+1);x.lineTo(bx+pad+ww*.62,yy+wh-1);x.stroke();
        }
      }
    }
    /* Structural vertical mullions vary in strength by facade system. */
    if(col>0){
      x.strokeStyle=facade==='glass-bands'?'rgba(21,49,67,.56)':'rgba(25,48,64,.38)';
      x.lineWidth=facade==='glass-bands'?1.05:.72;
      x.beginPath();x.moveTo(bx,top+3);x.lineTo(bx,bottom);x.stroke();
      x.strokeStyle='rgba(235,248,247,.16)';x.lineWidth=.55;
      x.beginPath();x.moveTo(bx+1.3,top+4);x.lineTo(bx+1.3,bottom);x.stroke();
    }
  }
  /* Fine floor plates, projecting ledges and believable shadow breaks. */
  if(masonry||facade==='vertical-bays'){
    for(let row=1;row<floors;row++){
      const yy=top+row*(h-8)/floors;
      x.fillStyle='rgba(24,46,60,.23)';x.fillRect(cx+1,yy,front-2,1.25);
      x.fillStyle='rgba(230,242,237,.2)';x.fillRect(cx+1,yy+1.25,front-2,.65);
    }
  }

  const refl=x.createLinearGradient(cx+front*.08,top,cx+front*.72,bottom);
  refl.addColorStop(0,'rgba(255,255,255,0)');
  refl.addColorStop(.36,'rgba(226,244,247,.44)');
  refl.addColorStop(.52,'rgba(245,250,244,.16)');
  refl.addColorStop(1,'rgba(255,255,255,0)');
  x.fillStyle=refl;x.beginPath();x.moveTo(cx+front*.08,top);x.lineTo(cx+front*.24,top);x.lineTo(cx+front*.62,bottom);x.lineTo(cx+front*.42,bottom);x.closePath();x.fill();
  x.restore();
  /* Dark edge at the street-facing corner gives each tower a strong 3-D read. */
  x.fillStyle='rgba(12,31,47,.2)';x.fillRect(right-1.2,top+4,1.2,h-8);
  x.restore();
}

  ctx.beginPath();ctx.ellipse(0,-43,14,17,-.12,0,Math.PI*2);ctx.fill();ctx.stroke();
  ctx.fillStyle='#F5FBFF';
  ctx.beginPath();ctx.ellipse(-6,-45,4.2,2.6,-.35,0,Math.PI*2);ctx.fill();
  ctx.beginPath();ctx.ellipse(6,-45,4.2,2.6,.35,0,Math.PI*2);ctx.fill();
  ctx.restore();
  /* Landing ripple gives the dive a readable destination near the street. */
  if(p>.78){
    const q=(p-.78)/.22;
    ctx.globalAlpha=fade*(1-q)*.48;ctx.strokeStyle='#F6D5C7';ctx.lineWidth=1.4;
    ctx.beginPath();ctx.ellipse(x,H*.91,W*(.035+q*.035),H*(.009+q*.008),0,0,Math.PI*2);ctx.stroke();
  }
  ctx.restore();
}

function render(e){
  ctx.setTransform(dpr,0,0,dpr,0,0);
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';
  if(sky)ctx.drawImage(sky,0,0,W,H);
  drawClouds();
  /* Far-side Manhattan skyline is smaller and softened by distance. */
  const dive=1.018+.007*(.5+.5*Math.sin(t*.22))+e*.002;
  const drift=2.5+3*Math.sin(t*.035)+e*3+ptr.y*2;
  const cityScale=(H*.60)/(H+EXTRA);
  ctx.save();ctx.translate(W*.5+ptr.x*W*.004,H*.60+ptr.y*H*.003);ctx.scale(dive,dive);ctx.translate(-W*.5,-H*.60);
  layers.forEach((L,i)=>{
    /* Scrolling down moves each building layer DOWN at a different depth rate. */
    const scrollShift=H*.14*sceneProgress()*(.16+L.pf*.72);
    const inkFrame=Math.floor(t*12)/12;
    const idleDrift=Math.sin(inkFrame*.29+i*.82)*(1.6+i*1.1);
    const shift=scrollShift+drift*L.depth*.26+idleDrift;
    /* Horizontal pointer parallax: near layers travel farther than distant layers. */
    const sideParallax=ptr.x*L.depth*W*(isSmall()?.026:.042);
    const dy=H*.075+shift,x0=-L.off+sideParallax;
    ctx.globalAlpha=L.alpha*.92;
    for(let xx=x0-L.tw;xx<W+L.tw;xx+=L.tw)ctx.drawImage(L.c,xx,dy,L.tw,L.hh*cityScale);
    if(i===0&&!reduced)drawFlicker(L,dy,x0,cityScale);
  });
  drawAnimeSunSweep(e);
  ctx.restore();ctx.globalAlpha=1;
  /* Soft bloom catches the skyline edges without flattening the building silhouettes. */
  drawGlows();
  /* The broad lake sits in front of the far shore, clearly separating both city layers. */
  drawLake();
  drawNearShoreCity();
  drawStreet();
  drawHeroLeap(e);
  /* Atmospheric particles, light trails and fine comic-print grain add cinematic life. */
  drawTrails();
  drawParts();
  drawHalftone(e);
  drawComicMotionPass(e);
  drawCinematicFinish(e);
  ctx.globalAlpha=1;ctx.globalCompositeOperation='source-over';
}

function drawAnimeSunSweep(e){
  if(!W||!H)return;
  const bandY=H*.40, bandH=H*.60;
  const sweep=((t*.075)%1.8)-.4;
  ctx.save();
  ctx.beginPath();ctx.rect(0,bandY,W,bandH);ctx.clip();
  ctx.globalCompositeOperation='screen';
  ctx.globalAlpha=.07+.025*Math.sin(t*.45);
  const x=sweep*W;
  const g=ctx.createLinearGradient(x-W*.22,bandY,x+W*.32,bandY+bandH*.55);
  g.addColorStop(0,'rgba(255,255,255,0)');
  g.addColorStop(.38,'rgba(255,225,168,.025)');
  g.addColorStop(.5,'rgba(255,249,220,.18)');
  g.addColorStop(.62,'rgba(255,218,154,.07)');
  g.addColorStop(1,'rgba(255,255,255,0)');
  ctx.fillStyle=g;ctx.fillRect(0,bandY,W,bandH);
  /* Thin specular streaks make glass facades feel alive, not like a flat texture. */
  ctx.globalAlpha=.065+.025*Math.sin(t*.75);
  ctx.translate(x,bandY);
  ctx.rotate(-.28);
  for(let i=0;i<7;i++){
    const yy=i*H*.105;
    const streak=ctx.createLinearGradient(-W*.5,yy,W*.5,yy);
    streak.addColorStop(0,'rgba(255,255,255,0)');
    streak.addColorStop(.48,'rgba(255,244,207,.5)');
    streak.addColorStop(1,'rgba(255,255,255,0)');
    ctx.fillStyle=streak;ctx.fillRect(-W*.7,yy,W*1.4,1.2+(i%3)*.7);
  }
  ctx.restore();
}
function updateClouds(dt){
  for(const c of clouds){c.x+=c.vx*dt;if(c.x>W+80)c.x=-c.w-rnd(40,200)}
}
function updateTrails(dt){
  for(let i=trails.length-1;i>=0;i--){
    const tr=trails[i];
    tr.x+=Math.cos(tr.a)*tr.spd*dt;tr.y+=Math.sin(tr.a)*tr.spd*dt;tr.life-=dt;
    if(tr.life<=0||tr.x<-400||tr.x>W+400||tr.y<-400||tr.y>H+400)trails.splice(i,1);
  }
}
function updateParts(dt){
  for(let i=parts.length-1;i>=0;i--){
    const p=parts[i];
    p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;
    if(p.life!==Infinity){p.life-=dt;if(p.life<=0){parts.splice(i,1);continue}}
    if(p.x<-12)p.x+=W+24;else if(p.x>W+12)p.x-=W+24;
    if(p.y<-12)p.y+=H+24;else if(p.y>H+12)p.y-=H+24;
  }
}

function step(dt){
  let e=0;
  if(!reduced){
    t+=dt;
    ptr.x+=(ptr.tx-ptr.x)*Math.min(1,dt*4);
    ptr.y+=(ptr.ty-ptr.y)*Math.min(1,dt*4);
    scr.y+=(scr.target-scr.y)*Math.min(1,dt*6);
    leapClock+=dt;
    if(leapT<0&&leapClock>=nextLeap){
      leapT=0;nextLeap=leapClock+(isSmall()?22:16)*rnd(.9,1.15);
      for(let i=0;i<(isSmall()?6:14);i++)parts.push(burstPart());
      for(let i=0;i<3;i++)spawnTrail(true);
    }
    if(leapT>=0){leapT+=dt;const u=leapT/LEAP_DUR;if(u>=1)leapT=-1;else e=Math.sin(Math.PI*u)}
    spawnAcc+=(isSmall()?.25:.5)*(1+e*5)*dt;
    while(spawnAcc>=1){spawnAcc-=1;if(trails.length<(isSmall()?6:14))spawnTrail(e>.2)}
    updateClouds(dt);updateTrails(dt);updateParts(dt);
    for(const L of layers)L.off=(L.off+L.speed*(1+e*L.boost)*dt)%L.tw;
  }
  render(e);
}

function loop(now){
  raf=0;
  const minGap=1000/(isSmall()?24:30);
  if(now-last>=minGap-2){
    const dt=Math.min(.05,Math.max(0,(now-last)/1000));last=now;step(dt);
  }
  if(!document.hidden)raf=requestAnimationFrame(loop);
}
function start(){
  if(raf||reduced)return;
  last=performance.now();raf=requestAnimationFrame(loop);
}

function build(){
  W=Math.max(1,window.innerWidth);H=Math.max(1,window.innerHeight);
  dpr=Math.min(window.devicePixelRatio||1,isSmall()?1.15:1.5);
  Q=Math.min(dpr,1.5);
  cv.width=Math.round(W*dpr);cv.height=Math.round(H*dpr);
  ctx.setTransform(dpr,0,0,dpr,0,0);
  buildSky();buildGlows();buildClouds();halftone=makeHalftone();dotTile=makeDotTile();
  layers=LAYER_DEFS.map(d=>{const L=d.dense?buildDense(d):buildLayer(d);return Object.assign({off:Math.random()*L.tw},d,L)});
  const want=isSmall()?8:16;
  parts=parts.filter(p=>p.life!==Infinity);
  for(let i=0;i<want;i++)parts.push(ambientPart(true));
  if(reduced)render(0);else start();
}
